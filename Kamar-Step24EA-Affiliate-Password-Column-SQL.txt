-- KAMAR STEP 24EA — Affiliate Password Column
-- Jalankan di Supabase SQL Editor sebelum mengetes login affiliate.

begin;

alter table public.affiliates
  add column if not exists password_hash text;

create index if not exists affiliates_password_email_idx
on public.affiliates (lower(email));

grant select, insert, update, delete on public.affiliates to anon, authenticated;

commit;
